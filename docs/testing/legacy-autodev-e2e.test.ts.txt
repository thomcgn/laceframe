interface FakeProcess {
  name: string;
  args: string[];
  cwd: string;
  env: Record<string, string>;
  exitCode: number | null;
  signal: NodeJS.Signals | null;
  stdout: string[];
  stderr: string[];
  killed: boolean;
  pid: number;
}

interface TestFixture {
  repoDir: string;
  binDir: string;
  originDir: string;
  originalPath: string;
  originalEnv: NodeJS.ProcessEnv;
  fakeProcesses: FakeProcess[];
  controllerLockPath: string;

  checkpointFile: string;
  budgetFile: string;
  configFile: string;
  logDir: string;
  reviewsDir: string;
}

const FAKE_KILO_SCRIPT = `#!/usr/bin/env node
// Fake Kilo CLI for deterministic testing
const fs = require('fs');
const path = require('path');

const args = process.argv.slice(2);
const sessionIdx = args.indexOf('--session');
const sessionId = sessionIdx >= 0 ? args[sessionIdx + 1] : null;
const promptIdx = args.indexOf('run');
const prompt = promptIdx >= 0 ? args.slice(promptIdx + 1).join(' ') : '';

// Parse environment for behavior control
const behavior = process.env.FAKE_KILO_BEHAVIOR || 'success';
const outputDelay = parseInt(process.env.FAKE_KILO_OUTPUT_DELAY || '0', 10);
const heartbeatInterval = parseInt(process.env.FAKE_KILO_HEARTBEAT || '1000', 10);
const sessionIdOutput = process.env.FAKE_KILO_SESSION_ID || 'test-session-' + Date.now();

function writeOutput(stream, data) {
  stream.write(data);
  if (process.env.FAKE_KILO_LOG) {
    fs.appendFileSync(process.env.FAKE_KILO_LOG, '[' + new Date().toISOString() + '] [' + (stream === process.stdout ? 'stdout' : 'stderr') + '] ' + data);
  }
}

function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function main() {
  // Simulate session ID output on startup
  if (behavior !== 'no-session-id') {
    writeOutput(process.stdout, 'session: ' + sessionIdOutput + '\\n');
  }

  if (outputDelay > 0) {
    await sleep(outputDelay);
  }

  let heartbeatCount = 0;
  const maxHeartbeats = parseInt(process.env.FAKE_KILO_MAX_HEARTBEATS || '999999', 10);

  const heartbeatTimer = setInterval(() => {
    heartbeatCount++;
    if (behavior === 'heartbeat') {
      writeOutput(process.stdout, 'heartbeat ' + heartbeatCount + '\\n');
    }
    if (heartbeatCount >= maxHeartbeats) {
      clearInterval(heartbeatTimer);
    }
  }, heartbeatInterval);

  // Behavior-specific logic
  switch (behavior) {
    case 'success':
      await sleep(parseInt(process.env.FAKE_KILO_RUNTIME || '100', 10));
      writeOutput(process.stdout, 'Task completed successfully\\n');
      process.exit(0);
      break;

    case 'rate-limit':
      writeOutput(process.stderr, 'Error: Free model usage limit reached. Retry-After: 120\\n');
      process.exit(1);
      break;

    case 'rate-limit-retry-after':
      const retryAfter = process.env.FAKE_KILO_RETRY_AFTER || '120';
      writeOutput(process.stderr, 'Error: Rate limited. Retry-After: ' + retryAfter + '\\n');
      process.exit(1);
      break;

    case 'server-error':
      writeOutput(process.stderr, 'Error: 503 Service Unavailable\\n');
      process.exit(1);
      break;

    case 'timeout':
      // Run forever until killed - simulates hanging process
      await new Promise(() => {}); // Never resolves
      break;

    case 'heartbeat':
      // Run with heartbeats until maxHeartbeats, then exit successfully
      await new Promise(() => {}); // Wait for signal
      break;

    case 'permission-rejected':
      writeOutput(process.stdout, 'run ended with an auto-rejected permission\\n');
      process.exit(1);
      break;

    case 'no-session-id':
      writeOutput(process.stdout, 'Task completed without session ID\\n');
      process.exit(0);
      break;

    default:
      writeOutput(process.stdout, 'Task completed\\n');
      process.exit(0);
  }
}

main().catch(err => {
  writeOutput(process.stderr, 'Fatal: ' + err.message + '\\n');
  process.exit(1);
});
`;

const FAKE_GH_SCRIPT = `#!/usr/bin/env node
// Fake GitHub CLI for deterministic testing
const fs = require('fs');
const path = require('path');

const args = process.argv.slice(2);
const behavior = process.env.FAKE_GH_BEHAVIOR || 'success';

function writeOutput(stream, data) {
  stream.write(data);
  if (process.env.FAKE_GH_LOG) {
    fs.appendFileSync(process.env.FAKE_GH_LOG, '[' + new Date().toISOString() + '] [' + (stream === process.stdout ? 'stdout' : 'stderr') + '] ' + data);
  }
}

function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

async function main() {
  // Simulate network delay
  await sleep(parseInt(process.env.FAKE_GH_DELAY || '0', 10));

  switch (behavior) {
    case 'success':
      if (args.includes('issue') && args.includes('list')) {
        const labelIdx = args.indexOf('--label');
        const label = labelIdx >= 0 ? args[labelIdx + 1] : '';
        if (label.includes('autodev:approved')) {
          // Return approved issues based on FAKE_GH_ISSUES env var
          const issues = JSON.parse(process.env.FAKE_GH_ISSUES || '[]');
          writeOutput(process.stdout, JSON.stringify(issues));
        } else {
          writeOutput(process.stdout, '[]');
        }
      } else if (args.includes('issue') && args.includes('view')) {
        const issueNum = args[args.indexOf('view') + 1];
        const issues = JSON.parse(process.env.FAKE_GH_ISSUES || '[]');
        const issue = issues.find(i => i.number == issueNum) || {
          number: parseInt(issueNum),
          title: 'Issue ' + issueNum,
          body: 'Implement behavior',
          labels: [{ name: 'autodev:approved' }],
          state: 'OPEN'
        };
        writeOutput(process.stdout, JSON.stringify(issue));
      } else if (args.includes('issue') && (args.includes('edit') || args.includes('comment') || args.includes('close'))) {
        // Success for mutations
        writeOutput(process.stdout, '');
      } else if (args.includes('auth') && args.includes('status')) {
        writeOutput(process.stdout, 'Authenticated');
      }
      process.exit(0);
      break;

    case 'auth-fail':
      if (args.includes('auth') && args.includes('status')) {
        writeOutput(process.stderr, 'Not authenticated\\n');
        process.exit(1);
      }
      process.exit(0);
      break;

    case 'network-error':
      writeOutput(process.stderr, 'Network error\\n');
      process.exit(1);
      break;

    case 'rate-limit':
      writeOutput(process.stderr, 'API rate limit exceeded\\n');
      process.exit(1);
      break;

    default:
      process.exit(0);
  }
}

main().catch(err => {
  writeOutput(process.stderr, 'Fatal: ' + err.message + '\\n');
  process.exit(1);
});
`;

function cleanupFixture(fixture: TestFixture): void {
  try {
    // Kill any remaining fake processes
    for (const proc of fixture.fakeProcesses) {
      if (!proc.killed && proc.exitCode === null) {
        try { process.kill(-proc.pid || 0, 'SIGKILL'); } catch { /* ignore */ }
      }
    }
  } catch { /* ignore */ }

  // Restore environment
  process.env.PATH = fixture.originalPath;
  for (const key of Object.keys(process.env)) {
    if (!Object.prototype.hasOwnProperty.call(fixture.originalEnv, key)) {
      delete process.env[key];
    }
  }
  for (const [key, value] of Object.entries(fixture.originalEnv)) {
    process.env[key] = value;
  }

  // Remove temp directories
  try { fs.rmSync(fixture.repoDir, { recursive: true, force: true }); } catch { /* ignore */ }
  try { fs.rmSync(fixture.binDir, { recursive: true, force: true }); } catch { /* ignore */ }
  try { fs.rmSync(fixture.originDir, { recursive: true, force: true }); } catch { /* ignore */ }
}

function setupEnvironment(fixture: TestFixture): void {
  // Prepend our fake bin directory to PATH
  process.env.PATH = `${fixture.binDir}:${fixture.originalPath}`;
  
  // Set AutoDev base directory to test fixture for isolation
  process.env.AUTODEV_BASE_DIR = fixture.repoDir;
  
  // Skip clean working tree check for E2E tests (fixture files are committed, but autodev files are created during test)
  process.env.AUTODEV_SKIP_CLEAN_CHECK = 'true';
  
  // Set up fake Kilo/GH logging
  process.env.FAKE_KILO_LOG = path.join(fixture.logDir, 'fake-kilo.log');
  process.env.FAKE_GH_LOG = path.join(fixture.logDir, 'fake-gh.log');
  
  // Ensure log directory exists for fake kilo
  fs.mkdirSync(path.dirname(process.env.FAKE_KILO_LOG), { recursive: true });
  
  // Default behaviors
  process.env.FAKE_KILO_BEHAVIOR = 'success';
  process.env.FAKE_GH_BEHAVIOR = 'success';
  process.env.FAKE_GH_ISSUES = JSON.stringify([{
    number: 39,
    title: "Test Issue",
    body: "Priority: high\nDependencies: \nRequired Validation: \nRelevant Files: ",
    labels: [{ name: "autodev:approved" }],
    state: "OPEN"
  }]);
}

function readJsonFile<T extends object>(filePath: string): T | null {
  try {
    return JSON.parse(fs.readFileSync(filePath, 'utf-8')) as T;
  } catch {
    return null;
  }
}

// --- Time Control for Tests ---

let timeOffset = 0;
const originalDateNow = Date.now;
const originalSetTimeout = global.setTimeout;
const originalClearTimeout = global.clearTimeout;
const originalSetInterval = global.setInterval;
const originalClearInterval = global.clearInterval;

function installTimeControl(): void {
  timeOffset = 0;
  global.Date.now = () => originalDateNow() + timeOffset;
  
  global.setTimeout = ((callback: () => void, delay: number, ...args: unknown[]) => {
    return originalSetTimeout(callback, Math.max(1, delay), ...args);
  }) as typeof setTimeout;
  
  global.setInterval = ((callback: () => void, delay: number, ...args: unknown[]) => {
    return originalSetInterval(callback, Math.max(1, delay), ...args);
  }) as typeof setInterval;
}

function uninstallTimeControl(): void {
  global.Date.now = originalDateNow;
  global.setTimeout = originalSetTimeout;
  global.clearTimeout = originalClearTimeout;
  global.setInterval = originalSetInterval;
  global.clearInterval = originalClearInterval;
}

// Helper to capture stdout/stderr
function captureOutput(): { stdout: string; stderr: string; restore: () => void } {
  let stdout = '';
  let stderr = '';
  const originalStdoutWrite = process.stdout.write.bind(process.stdout);
  const originalStderrWrite = process.stderr.write.bind(process.stderr);
  process.stdout.write = (chunk: string) => { stdout += chunk; return originalStdoutWrite(chunk); };
  process.stderr.write = (chunk: string) => { stderr += chunk; return originalStderrWrite(chunk); };
  return {
    get stdout() { return stdout; },
    get stderr() { return stderr; },
    restore: () => {
      process.stdout.write = originalStdoutWrite;
      process.stderr.write = originalStderrWrite;
    }
  };
}

describe("AutoDev E2E Integration Tests (Issue #39)", () => {
  let fixture: TestFixture;

  beforeAll(() => {
    installTimeControl();
  });

  afterAll(() => {
    uninstallTimeControl();
  });

  beforeEach(() => {
    fixture = createTestFixture();
    setupEnvironment(fixture);
  });

  afterEach(async () => {
    // Ensure cleanup happens even if test fails
    cleanupFixture(fixture);
    vi.resetAllMocks();
  });

  // Helper to update config file for a test
  function updateConfig(fixture: TestFixture, config: Record<string, unknown>): void {
    const configPath = path.join(fixture.repoDir, '.autodev-config.json');
    const existing = JSON.parse(fs.readFileSync(configPath, 'utf-8'));
    const merged = { ...existing, ...config };
    fs.writeFileSync(configPath, JSON.stringify(merged, null, 2));
  }

  describe("Scenario: No fixed 30-minute limit with heartbeats", () => {
    it("allows simulated runtime >30 minutes with continuous output/heartbeats without abort", async () => {
      // Configure fake Kilo to emit heartbeats for extended period
      process.env.FAKE_KILO_BEHAVIOR = 'heartbeat';
      process.env.FAKE_KILO_HEARTBEAT = '100'; // Fast heartbeats for test
      process.env.FAKE_KILO_MAX_HEARTBEATS = '500'; // Simulate >30 min equivalent
      process.env.FAKE_KILO_RUNTIME = '5000'; // 5 seconds test time

      updateConfig(fixture, {
        kilo: { inactivityTimeoutMs: 30000, sessionResume: true }
      });

      const { runAutodev } = await import("../src/autodev/controller.js");
      
      const output = captureOutput();

      try {
        // This should not be terminated by inactivity because heartbeats keep coming
        await Promise.race([
          runAutodev(false),
          new Promise((_, reject) => setTimeout(() => reject(new Error('Test timeout')), 15000))
        ]);
      } catch {
        // Controller may exit after processing one issue in dry-run mode
        // The key assertion is that inactivity warning fires but process continues
      } finally {
        output.restore();
      }

      // Verify inactivity warning was logged (at 80% of timeout)
      expect(output.stderr).toContain('INACTIVITY WARNING');
      
      // Verify process was not killed due to inactivity
      const kiloLog = fs.readFileSync(process.env.FAKE_KILO_LOG!, 'utf-8');
      expect(kiloLog).toContain('heartbeat');
    }, 20000);
  });

  describe("Scenario: Inactivity abort with SIGTERM/SIGKILL", () => {
    it("terminates silent process with SIGTERM then SIGKILL, persists checkpoint", async () => {
      // Configure fake Kilo to produce NO output (silence)
      process.env.FAKE_KILO_BEHAVIOR = 'timeout'; // Runs forever silently
      process.env.FAKE_KILO_OUTPUT_DELAY = '0';

      updateConfig(fixture, {
        kilo: { inactivityTimeoutMs: 1000, sessionResume: true }
      });

      const { runAutodev } = await import("../src/autodev/controller.js");
      
      const output = captureOutput();

      try {
        await Promise.race([
          runAutodev(false),
          new Promise((_, reject) => setTimeout(() => reject(new Error('Test timeout')), 10000))
        ]);
      } catch {
        // Expected: controller should be terminated by inactivity
      } finally {
        output.restore();
      }

      // Verify checkpoint was persisted
      const checkpoint = readJsonFile<{ phase: string; reason: string }>(fixture.checkpointFile);
      expect(checkpoint).not.toBeNull();
      expect(checkpoint?.phase).toBe('builder');
      expect(checkpoint?.reason).toBe('inactivity');
      
      // Verify inactivity timeout was logged
      expect(output.stderr).toContain('Inactivity timeout reached');
    }, 15000);
  });

  describe("Scenario: 429 / Rate Limit with Retry-After", () => {
    it("handles 429 with Retry-After, persists cooldown and checkpoint, blocks new builder until expiry", async () => {
      // First run: Kilo returns 429 with Retry-After
      process.env.FAKE_KILO_BEHAVIOR = 'rate-limit-retry-after';
      process.env.FAKE_KILO_RETRY_AFTER = '1'; // 1 second for test
      process.env.FAKE_KILO_OUTPUT_DELAY = '100';

      updateConfig(fixture, {
        kilo: { inactivityTimeoutMs: 30000, sessionResume: true },
        freeBudget: { cooldownMinutes: 0, windowMinutes: 60, limit: 200, warningAt: 160, pauseAt: 190, estimatedRequestsPerTask: 50 }
      });

      const { runAutodev } = await import("../src/autodev/controller.js");
      
      const output1 = captureOutput();

      try {
        await Promise.race([
          runAutodev(false),
          new Promise((_, reject) => setTimeout(() => reject(new Error('Test timeout')), 10000))
        ]);
      } catch {
        // Expected to fail due to rate limit
      } finally {
        output1.restore();
      }

      // Verify checkpoint was saved with rate-limit reason
      const checkpoint = readJsonFile<{ reason: string }>(fixture.checkpointFile);
      expect(checkpoint).not.toBeNull();
      expect(checkpoint?.reason).toBe('rate-limit');

      // Verify budget file has cooldown
      const budget = readJsonFile<{ cooldownUntil: number }>(fixture.budgetFile);
      expect(budget).not.toBeNull();
      expect(budget?.cooldownUntil).toBeGreaterThan(Date.now());

      // Second run (simulated restart): should detect cooldown and NOT start new builder
      process.env.FAKE_KILO_BEHAVIOR = 'success';
      process.env.FAKE_KILO_OUTPUT_DELAY = '100';

      const output2 = captureOutput();

      try {
        await Promise.race([
          runAutodev(false),
          new Promise((_, reject) => setTimeout(() => reject(new Error('Test timeout')), 10000))
        ]);
      } catch {
        // Expected to wait
      } finally {
        output2.restore();
      }

      // Should log that it's waiting for budget / cooldown
      expect(output2.stderr).toContain('paused');
      expect(output2.stderr).toContain('Resuming no earlier than');
    }, 20000);

    it("continues controlled after cooldown expires", async () => {
      // Set up expired cooldown in budget file
      const expiredBudget = {
        version: 1,
        reservations: [],
        cooldownUntil: Date.now() - 1000, // Already expired
        pauseReason: 'rate limit hit'
      };
      fs.writeFileSync(fixture.budgetFile, JSON.stringify(expiredBudget));

      process.env.FAKE_KILO_BEHAVIOR = 'success';
      process.env.FAKE_KILO_OUTPUT_DELAY = '100';

      const { runAutodev } = await import("../src/autodev/controller.js");

      const output = captureOutput();

      try {
        await Promise.race([
          runAutodev(false),
          new Promise((_, reject) => setTimeout(() => reject(new Error('Test timeout')), 10000))
        ]);
      } catch {
        // May succeed or fail depending on other factors
      } finally {
        output.restore();
      }

      // Should proceed without waiting
      expect(output.stderr).not.toContain('paused');
    }, 15000);
  });

  describe("Scenario: 503 Transient Error", () => {
    it("handles 503 without blind retry when working tree modified", async () => {
      updateConfig(fixture, {
        kilo: { inactivityTimeoutMs: 30000, sessionResume: true }
      });
      process.env.FAKE_KILO_BEHAVIOR = 'server-error'; // 503
      process.env.FAKE_KILO_OUTPUT_DELAY = '100';

      const { runAutodev } = await import("../src/autodev/controller.js");

      const output = captureOutput();

      try {
        await Promise.race([
          runAutodev(false),
          new Promise((_, reject) => setTimeout(() => reject(new Error('Test timeout')), 10000))
        ]);
      } catch {
        // Expected to fail
      } finally {
        output.restore();
      }

      // Should fail the build, not retry blindly
      // Controller should handle infrastructure error
      const hasError = output.stderr.includes('503') || output.stderr.includes('Service Unavailable') || output.stderr.includes('KiloExecutionError');
      expect(hasError).toBe(true);
    }, 15000);
  });

  describe("Scenario: Session Resume", () => {
    it("saves session ID from Kilo output and uses --session on restart", async () => {
      updateConfig(fixture, {
        kilo: { inactivityTimeoutMs: 30000, sessionResume: true }
      });
      // First run: Kilo outputs session ID
      process.env.FAKE_KILO_BEHAVIOR = 'success';
      process.env.FAKE_KILO_SESSION_ID = 'test-session-resume-123';
      process.env.FAKE_KILO_OUTPUT_DELAY = '100';

      const { runAutodev } = await import("../src/autodev/controller.js");

      const output1 = captureOutput();

      try {

describe("Kilo Runner E2E (real spawn/signal)", () => {
  let fixture: TestFixture;

  beforeAll(() => {
    installTimeControl();
  });

  afterAll(() => {
    uninstallTimeControl();
  });

  beforeEach(() => {
    fixture = createTestFixture();
    setupEnvironment(fixture);
  });

  afterEach(() => {
    cleanupFixture(fixture);
    vi.resetAllMocks();
  });

  it("spawns kilo with correct arguments including --session", async () => {
    process.env.FAKE_KILO_BEHAVIOR = 'success';
    process.env.FAKE_KILO_SESSION_ID = 'runner-session-123';
    process.env.FAKE_KILO_OUTPUT_DELAY = '50';

    const { runKiloStreaming } = await import("../src/autodev/kilo-runner.js");

    const result = await runKiloStreaming({
      agent: 'laceframe-builder',
      model: 'test-model',
      prompt: 'test prompt',
      logPath: path.join(fixture.logDir, 'kilo-runner.log'),
      inactivityTimeoutMs: 30000,
      sessionId: 'runner-session-123',
      onOutput: vi.fn(),
      onHeartbeat: vi.fn(),
      onInactivityWarning: vi.fn(),
      registerCancel: vi.fn(),
    });

    expect(result.sessionId).toBe('runner-session-123');
    expect(result.terminatedByInactivity).toBe(false);
    
    // Verify the spawned command included --session
    const logContent = fs.readFileSync(process.env.FAKE_KILO_LOG!, 'utf-8');
    expect(logContent).toContain('session: runner-session-123');
  }, 10000);

  it("terminates on inactivity timeout with SIGTERM then SIGKILL", async () => {
    process.env.FAKE_KILO_BEHAVIOR = 'timeout'; // Silent, runs forever
    process.env.FAKE_KILO_OUTPUT_DELAY = '0';

    const { runKiloStreaming } = await import("../src/autodev/kilo-runner.js");

    const killCalls: { pid: number; signal: string }[] = [];
    const originalKill = process.kill;
    // Wrap process.kill to track calls but still perform the actual kill
    process.kill = vi.fn((pid: number, signal?: string | number) => {
      killCalls.push({ pid, signal: String(signal || 'SIGTERM') });
      return originalKill(pid, signal);
    }) as unknown as typeof process.kill;

    try {
      await runKiloStreaming({
        agent: 'test',
        model: 'test',
        prompt: 'test',
        logPath: path.join(fixture.logDir, 'kilo-timeout.log'),
        inactivityTimeoutMs: 500, // Very short for test
        heartbeatIntervalMs: 100, // Fast heartbeat for test
        onOutput: vi.fn(),
        onHeartbeat: vi.fn(),
        onInactivityWarning: vi.fn(),
        registerCancel: vi.fn(),
      });
    } catch (error) {
      // Expected to be rejected due to inactivity timeout
      const errMsg = String(error);
      expect(errMsg).toContain('terminated by inactivity');
      expect(errMsg).toContain('SIGTERM');
    } finally {
      process.kill = originalKill;
    }

    // Verify SIGTERM then SIGKILL sequence
    expect(killCalls.length).toBeGreaterThanOrEqual(1);
    expect(killCalls[0].signal).toBe('SIGTERM');
    expect(killCalls[0].pid).toBeLessThan(0); // Process group
    
    // If SIGKILL was called, it should be after SIGTERM
    const sigkillCall = killCalls.find(c => c.signal === 'SIGKILL');
    if (sigkillCall) {
      expect(killCalls.indexOf(sigkillCall)).toBeGreaterThan(0);
    }
  }, 10000);

  it("registers cancel function that kills process group", async () => {
    process.env.FAKE_KILO_BEHAVIOR = 'heartbeat';
    process.env.FAKE_KILO_HEARTBEAT = '100';
    process.env.FAKE_KILO_MAX_HEARTBEATS = '1000';

    const { runKiloStreaming } = await import("../src/autodev/kilo-runner.js");

    let cancelFn: (() => void) | undefined;
    let killCalled = false;
    let killPid = 0;
    let killSignal = '';
    
    const originalKill = process.kill;
    process.kill = vi.fn((pid: number, signal?: string | number) => {
      killCalled = true;
      killPid = pid;
      killSignal = String(signal || 'SIGTERM');
      return true;
    }) as unknown as typeof process.kill;

    const promise = runKiloStreaming({
      agent: 'test',
      model: 'test',
      prompt: 'test',
      logPath: path.join(fixture.logDir, 'kilo-cancel.log'),
      inactivityTimeoutMs: 30000,
      heartbeatIntervalMs: 100, // Fast heartbeat for test
      onOutput: vi.fn(),
      onHeartbeat: vi.fn(),
      onInactivityWarning: vi.fn(),
      registerCancel: (cancel) => { cancelFn = cancel; },
    });

    // Wait a bit for spawn
    await new Promise(r => setTimeout(r, 100));
    
    // Call cancel
    expect(cancelFn).toBeDefined();
    cancelFn!();
    
    // Wait for process to close
    try {
      await Promise.race([promise, new Promise((_, r) => setTimeout(() => r(new Error('timeout')), 2000))]);
    } catch {
      // Ignore timeout
    }

    process.kill = originalKill;

    expect(killCalled).toBe(true);
    expect(killSignal).toBe('SIGTERM');
    expect(killPid).toBeLessThan(0); // Process group
  }, 10000);
});
